### 5.1 `src/lib/import/sourceReviews.ts`（新檔，照抄；client／server 都會用，不准 import server 專用模組）
```ts
/** 4a: buyer-review summary the extension read from the source page (raw_capture.payload.source_reviews). */
export type SourceReviews = {
  source: "ice" | "dom";
  total_text: string | null;
  good_rate_text: string | null;
  tags: Array<{ text: string; count: number | null }>;
  samples: Array<{ text: string; sku: string | null; date: string | null }>;
  signals: string[];
};

export const SOURCE_REVIEW_LIMITS = { tags: 12, samples: 5, sampleChars: 300, signals: 6 };

function isPlainObject(value: unknown): value is Record<string, unknown> {
  return !!value && typeof value === "object" && !Array.isArray(value);
}

function text(value: unknown, max: number): string {
  return typeof value === "string" ? value.replace(/\s+/g, " ").trim().slice(0, max) : "";
}

/** Whitelists the 6 known keys, applies the limits, drops buyer identity; null when nothing is left. */
export function normalizeSourceReviews(value: unknown): SourceReviews | null {
  if (!isPlainObject(value) || (value.source !== "ice" && value.source !== "dom")) return null;
  const list = (v: unknown) => (Array.isArray(v) ? v : []);
  const tags = list(value.tags)
    .filter(isPlainObject)
    .map((t) => ({ text: text(t.text, 40), count: typeof t.count === "number" && Number.isFinite(t.count) ? t.count : null }))
    .filter((t) => t.text)
    .slice(0, SOURCE_REVIEW_LIMITS.tags);
  const samples = list(value.samples)
    .filter(isPlainObject)
    .map((s) => ({ text: text(s.text, SOURCE_REVIEW_LIMITS.sampleChars), sku: text(s.sku, 80) || null, date: text(s.date, 20) || null }))
    .filter((s) => s.text)
    .slice(0, SOURCE_REVIEW_LIMITS.samples);
  const signals = list(value.signals).map((s) => text(s, 40)).filter(Boolean).slice(0, SOURCE_REVIEW_LIMITS.signals);
  const out: SourceReviews = {
    source: value.source,
    total_text: text(value.total_text, 20) || null,
    good_rate_text: text(value.good_rate_text, 40) || null,
    tags,
    samples,
    signals
  };
  if (!out.total_text && !out.good_rate_text && !tags.length && !samples.length && !signals.length) return null;
  return out;
}

export function readSourceReviews(rawCapture: unknown): SourceReviews | null {
  if (!isPlainObject(rawCapture) || !isPlainObject(rawCapture.payload)) return null;
  return normalizeSourceReviews(rawCapture.payload.source_reviews);
}
```

### 5.2 `mapCaptureFields.ts`：拿掉「每款都一樣」的款式軸
**規則**（在 `mapVariantRows` 裡、`const cleaned = cleanVariantRows(rows);` 之後、`rows.forEach((row, i) => { row.source_options = …` 之前）：
- 只在保留的款式 ≥ 2 款時才判斷。
- 某個軸（第 1／2／3 軸）要被拿掉，必須**同時**符合：
  1. 每一款在這個軸都有值（沒有 null）；
  2. 所有保留款式＋「來源售完被排除」的款式（`excluded` 裡 `reason === "source_sold_out"`，值先過 `cleanOptionValue`）在這個軸的值**全部相同**；
  3. 另外還有一個**沒被拿掉**的軸有 ≥ 2 個不同值。
- 拿掉後把剩下的軸往前補（例：拿掉第 1 軸 → 原第 2 軸變第 1 軸），空出來的填 `null`。`cleaned` 和「來源售完被排除」款式的 `row.option*_name/value` 都要一起補位（不然之後「加回」會多出一個軸）。
- `source_options`（`names`／`values`）**照舊用拿掉前的原始資料**，不准改（STOCK／重抓用 `source_sku_id` 對款式，靠它）。
- `mapVariantRows` 多回傳 `droppedAxes: Array<{ name: string; value: string }>`（name／value 用清理後的文字，例 `全新盒裝`／`正版授權`），回傳型別要跟著加。

## 6. 店家共用宣傳圖
### 6.1 `src/lib/import/captureSharedImages.ts`（新檔）
規則：一張詳情圖如果**已經出現在 ≥ 2 件「其他商品」**的詳情圖裡 → 店家共用宣傳圖 → 排除。
- 「同一張圖」用 `detailImageKey`：網址裡有 `O1CN` 開頭的阿里圖片編號就用它（`/O1CN[0-9A-Za-z]+/`，不含後面的 `_`、`~`、`!!`）；沒有就用「小寫主機＋路徑」，去掉 `_400x400`／`_q90` 尺寸後綴和 `_.webp` 尾巴。
- 「其他商品」用 `extractUrlMatchKey`（`src/lib/drafts/checkDuplicate.ts`）比對商品；跟這次同一件、或 key 是空的不算；同一件其他商品出現多次只算 1 件。
- **保險**：要排除的張數 > 詳情圖總數的一半 → 一張都不排除，回傳 `skipped: "too_many"`（同系列商品可能共用整套詳情圖）。
- 歷史資料讀不到（查詢出錯）→ 不排除，照舊匯入。
```ts
import { extractUrlMatchKey } from "@/lib/drafts/checkDuplicate";

export const SHARED_IMAGE_MIN_OTHER_ITEMS = 2;
/** Newest drafts scanned for shared images (raw_capture.payload.detail_image_urls only). */
export const SHARED_IMAGE_HISTORY_LIMIT = 300;

export type ExcludedDetailImage = { url: string; key: string; reason: "shop_shared"; other_items: number };
export type DetailImageHistoryRow = { source_url: string | null; detail_image_urls: unknown };

export function detailImageKey(url: string): string | null {
  const id = String(url || "").match(/O1CN[0-9A-Za-z]+/);
  if (id) return id[0];
  try {
    const u = new URL(url);
    const path = u.pathname
      .replace(/\.(jpg|jpeg|png|webp)_\.webp$/i, ".$1")
      .replace(/_\.webp$/i, "")
      .replace(/_\d+x\d+(?:q\d+)?(\.\w+)?$/i, "$1")
      .replace(/_q\d+(\.\w+)?$/i, "$1");
    return u.hostname.toLowerCase() + path;
  } catch {
    return null;
  }
}

export function pickSharedDetailImages(input: {
  detailImageUrls: string[];
  sourceUrl: string;
  history: DetailImageHistoryRow[];
}): { kept: string[]; excluded: ExcludedDetailImage[]; skipped: "too_many" | null } {
  const current = extractUrlMatchKey(input.sourceUrl || "");
  const itemsByKey = new Map<string, Set<string>>();
  for (const row of input.history) {
    if (!row.source_url || !Array.isArray(row.detail_image_urls)) continue;
    const item = extractUrlMatchKey(row.source_url);
    if (!item || item === current) continue;
    for (const url of row.detail_image_urls) {
      const key = typeof url === "string" ? detailImageKey(url) : null;
      if (!key) continue;
      if (!itemsByKey.has(key)) itemsByKey.set(key, new Set());
      itemsByKey.get(key)!.add(item);
    }
  }
  const kept: string[] = [];
  const excluded: ExcludedDetailImage[] = [];
  for (const url of input.detailImageUrls) {
    const key = detailImageKey(url);
    const otherItems = key ? itemsByKey.get(key)?.size ?? 0 : 0;
    if (key && otherItems >= SHARED_IMAGE_MIN_OTHER_ITEMS) {
      excluded.push({ url, key, reason: "shop_shared", other_items: otherItems });
    } else {
      kept.push(url);
    }
  }
  if (excluded.length * 2 > input.detailImageUrls.length) {
    return { kept: [...input.detailImageUrls], excluded: [], skipped: "too_many" };
  }
  return { kept, excluded, skipped: null };
}

export function formatSharedImagesExcludedWarning(count: number): string {

export async function loadDetailImageHistory(db: any): Promise<DetailImageHistoryRow[] | null> {
  try {
    const { data, error } = await db
      .from("product_drafts")
      .select("source_url, detail_image_urls:raw_capture->payload->detail_image_urls")
      .not("raw_capture", "is", null)
      .order("created_at", { ascending: false })
      .limit(SHARED_IMAGE_HISTORY_LIMIT);
    if (error || !Array.isArray(data)) return null;
    return data as DetailImageHistoryRow[];
  } catch {
    return null;
  }
```

## 7. 加回被排除的圖
### 7.1 `src/lib/images/restoreExcludedDetailImage.ts`（新檔，照抄）
```ts
import type { ExcludedDetailImage } from "@/lib/import/captureSharedImages";

function isPlainObject(value: unknown): value is Record<string, unknown> {
  return !!value && typeof value === "object" && !Array.isArray(value);
}

/** 4a: detail images the capture import kept out as shop-shared promos (raw_capture.server.excluded_detail_images). */
export function readExcludedDetailImages(rawCapture: unknown): ExcludedDetailImage[] {
  if (!isPlainObject(rawCapture) || !isPlainObject(rawCapture.server)) return [];
  const list = rawCapture.server.excluded_detail_images;
  if (!Array.isArray(list)) return [];
  return list.filter(
    (e): e is ExcludedDetailImage => isPlainObject(e) && typeof e.url === "string" && typeof e.key === "string"
  );
}

export function planRestoreExcludedDetailImage(
  rawCapture: unknown,
  input: { index: number; url: string; published: boolean }
):
  | { ok: true; entry: ExcludedDetailImage; nextServer: Record<string, unknown> }
  | { ok: false; status: 404 | 409; error: string } {
  if (input.published) return { ok: false, status: 409, error: "已上架的商品請到 Shopify 後台加圖" };
  const list = readExcludedDetailImages(rawCapture);
  const entry = list[input.index];
  if (!entry || entry.url !== input.url) {
    return { ok: false, status: 404, error: "找不到這張已排除的圖（可能已經加回）" };
  }
  const server = isPlainObject(rawCapture) && isPlainObject(rawCapture.server) ? rawCapture.server : {};
  return { ok: true, entry, nextServer: { ...server, excluded_detail_images: list.filter((_, i) => i !== input.index) } };
}
```
